SELECT 
    CONCAT_WS(' ', s.cliente, NULLIF(s.apellidos, '')) AS nombre_completo,
    s.numero_de_caso,
    s.fecha_nacimiento,
    s.telefono,
    s.direccion,
    s.terapeuta AS id_terapeuta,
    u.nombre AS nombre_terapeuta,
    s.id_solicitud,
    s.num_sesiones AS horas_aprobadas,
    
    -- Fechas de las citas en el mes específico
    GROUP_CONCAT(
        CASE 
            WHEN YEAR(nc.fecha_creacion) = :anio AND MONTH(nc.fecha_creacion) = :mes 
            THEN DATE_FORMAT(nc.fecha_creacion, '%Y-%m-%d')
            ELSE NULL 
        END 
        SEPARATOR ', '
    ) AS fechas_citas_mes,
    
    -- Tipos de duración de sesión en el mes (ej: "1 & 2")
    GROUP_CONCAT(
        DISTINCT CASE 
            WHEN YEAR(nc.fecha_creacion) = :anio AND MONTH(nc.fecha_creacion) = :mes THEN
                CASE 
                    WHEN nc.tiempo_sesion <= 44 THEN '0.5'
                    WHEN nc.tiempo_sesion BETWEEN 45 AND 74 THEN '1'
                    WHEN nc.tiempo_sesion BETWEEN 75 AND 104 THEN '2'
                    WHEN nc.tiempo_sesion >= 105 THEN '3'
                END
            ELSE NULL
        END
        ORDER BY nc.tiempo_sesion ASC
        SEPARATOR ' & '
    ) AS duracion_sesiones_mes,

    -- Cantidad de sesiones/citas del mes específico
    SUM(
        CASE 
            WHEN YEAR(nc.fecha_creacion) = :anio AND MONTH(nc.fecha_creacion) = :mes THEN
                IF(c.dos_citas = 1, 2, 1)
            ELSE 0 
        END
    ) AS num_sesiones_mes,

    -- Total de horas del mes específico (según minutos * c.dos_citas)
    SUM(
        CASE 
            WHEN YEAR(nc.fecha_creacion) = :anio AND MONTH(nc.fecha_creacion) = :mes THEN
                (CASE 
                    WHEN nc.tiempo_sesion <= 44 THEN 0.5
                    WHEN nc.tiempo_sesion BETWEEN 45 AND 74 THEN 1.0
                    WHEN nc.tiempo_sesion BETWEEN 75 AND 104 THEN 2.0
                    WHEN nc.tiempo_sesion >= 105 THEN 3.0
                    ELSE 0
                END) * IF(c.dos_citas = 1, 2, 1)
            ELSE 0 
        END
    ) AS horas_del_mes,
    
    -- Total de horas consumidas en toda la solicitud (Histórico Mixto)
    SUM(
        CASE 
            -- Regla antigua: Hasta Mayo 2026 inclusive (Año < 2026 O (Año = 2026 y Mes <= 5))
            WHEN YEAR(nc.fecha_creacion) < 2026 
              OR (YEAR(nc.fecha_creacion) = 2026 AND MONTH(nc.fecha_creacion) <= 5) 
            THEN IF(c.dos_citas = 1, 2, 1)

            -- Regla nueva: A partir de Junio 2026 en adelante
            ELSE 
                (CASE 
                    WHEN nc.tiempo_sesion <= 44 THEN 0.5
                    WHEN nc.tiempo_sesion BETWEEN 45 AND 74 THEN 1.0
                    WHEN nc.tiempo_sesion BETWEEN 75 AND 104 THEN 2.0
                    WHEN nc.tiempo_sesion >= 105 THEN 3.0
                    ELSE 0
                END) * IF(c.dos_citas = 1, 2, 1)
        END
    ) AS total_horas_solicitud,
    
    -- Horas restantes del cliente (num_sesiones - total histórico mixto)
    (
        s.num_sesiones - SUM(
            CASE 
                -- Regla antigua: Hasta Mayo 2026 inclusive
                WHEN YEAR(nc.fecha_creacion) < 2026 
                  OR (YEAR(nc.fecha_creacion) = 2026 AND MONTH(nc.fecha_creacion) <= 5) 
                THEN IF(c.dos_citas = 1, 2, 1)

                -- Regla nueva: A partir de Junio 2026 en adelante
                ELSE 
                    (CASE 
                        WHEN nc.tiempo_sesion <= 44 THEN 0.5
                        WHEN nc.tiempo_sesion BETWEEN 45 AND 74 THEN 1.0
                        WHEN nc.tiempo_sesion BETWEEN 75 AND 104 THEN 2.0
                        WHEN nc.tiempo_sesion >= 105 THEN 3.0
                        ELSE 0
                    END) * IF(c.dos_citas = 1, 2, 1)
            END
        )
    ) AS horas_restantes,
    CASE s.sexo
        WHEN 'F' THEN 'Female'
        WHEN 'M' THEN 'Male'
        ELSE s.sexo -- Opcional: por si hay valores nulos o diferentes
    END AS sexo

FROM solicitud_voc s
LEFT JOIN usuario u ON s.terapeuta = u.id_usuario
LEFT JOIN cita c ON s.id_solicitud = c.id_solicitud
LEFT JOIN nota_cita nc ON c.id_cita = nc.id_cita

-- Filtro por ID de terapeuta (0 = todos los terapeutas)
WHERE (:idTerapeuta = 0 OR s.terapeuta = :idTerapeuta)

GROUP BY 
    s.cliente,
    s.numero_de_caso,
    s.fecha_nacimiento,
    s.telefono,
    s.direccion,
    s.terapeuta,
    u.nombre,
    s.id_solicitud, 
    s.num_sesiones

-- FILTRO POR CITAS DEL MES PARAMETRIZADO:
HAVING COUNT(
    CASE 
        WHEN YEAR(nc.fecha_creacion) = :anio AND MONTH(nc.fecha_creacion) = :mes 
        THEN nc.id_nota 
    END
) > 0;
