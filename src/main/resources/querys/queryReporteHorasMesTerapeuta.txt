SELECT 
    s.terapeuta,
    CONCAT_WS(' ', s.cliente, NULLIF(s.apellidos, '')) AS nombre_completo,
    s.id_solicitud,
    s.num_sesiones AS horas_aprobadas,
    
    -- Fechas de las citas en el mes específico
    GROUP_CONCAT(
        CASE 
            WHEN YEAR(nc.fecha_creacion) = :anio AND MONTH(nc.fecha_creacion) = :mes 
            THEN DATE_FORMAT(nc.fecha_creacion, '%Y-%m-%d')
            ELSE NULL 
        END 
        SEPARATOR ', '
    ) AS fechas_citas_mes,
    
    -- Tipos de duración de sesión en el mes (con la regla aplicada)
    GROUP_CONCAT(
        DISTINCT CASE 
            WHEN YEAR(nc.fecha_creacion) = :anio AND MONTH(nc.fecha_creacion) = :mes THEN
                CASE 
                    WHEN DATE(nc.fecha_creacion) <= '2026-07-15' THEN 
                        IF(c.dos_citas = 1, '2', '1')
                    ELSE 
                        CONCAT(
                            CASE 
                                WHEN nc.tiempo_sesion <= 44 THEN 0.5
                                WHEN nc.tiempo_sesion BETWEEN 45 AND 74 THEN 1.0
                                WHEN nc.tiempo_sesion BETWEEN 75 AND 89 THEN 1.5
                                WHEN nc.tiempo_sesion BETWEEN 90 AND 120 THEN 2.0
                            END * IF(c.dos_citas = 1, 2, 1)
                        )
                END
            ELSE NULL
        END
        ORDER BY nc.tiempo_sesion ASC
        SEPARATOR ' & '
    ) AS duracion_sesiones_mes,

    -- Total de horas consumidas en el mes específico
    SUM(
        CASE 
            WHEN YEAR(nc.fecha_creacion) = :anio AND MONTH(nc.fecha_creacion) = :mes THEN
                CASE 
                    WHEN DATE(nc.fecha_creacion) <= '2026-07-15' THEN 
                        IF(c.dos_citas = 1, 2, 1)
                    ELSE 
                        (CASE 
                            WHEN nc.tiempo_sesion <= 44 THEN 0.5
                            WHEN nc.tiempo_sesion BETWEEN 45 AND 74 THEN 1.0
                            WHEN nc.tiempo_sesion BETWEEN 75 AND 89 THEN 1.5
                            WHEN nc.tiempo_sesion BETWEEN 90 AND 120 THEN 2.0
                            ELSE 0
                        END) * IF(c.dos_citas = 1, 2, 1)
                END
            ELSE 0 
        END
    ) AS horas_del_mes,
    
    -- Total de horas consumidas en toda la solicitud (histórico global con regla nueva/antigua)
    SUM(
        CASE 
            WHEN DATE(nc.fecha_creacion) <= '2026-07-15' THEN 
                IF(c.dos_citas = 1, 2, 1)
            ELSE 
                (CASE 
                    WHEN nc.tiempo_sesion <= 44 THEN 0.5
                    WHEN nc.tiempo_sesion BETWEEN 45 AND 74 THEN 1.0
                    WHEN nc.tiempo_sesion BETWEEN 75 AND 89 THEN 1.5
                    WHEN nc.tiempo_sesion BETWEEN 90 AND 120 THEN 2.0
                    ELSE 0
                END) * IF(c.dos_citas = 1, 2, 1)
        END
    ) AS total_horas_solicitud,
    
    -- Horas restantes del cliente (num_sesiones - total histórico)
    (
        s.num_sesiones - SUM(
            CASE 
                WHEN DATE(nc.fecha_creacion) <= '2026-07-15' THEN 
                    IF(c.dos_citas = 1, 2, 1)
                ELSE 
                    (CASE 
                        WHEN nc.tiempo_sesion <= 44 THEN 0.5
                        WHEN nc.tiempo_sesion BETWEEN 45 AND 74 THEN 1.0
                        WHEN nc.tiempo_sesion BETWEEN 75 AND 89 THEN 1.5
                        WHEN nc.tiempo_sesion BETWEEN 90 AND 120 THEN 2.0
                        ELSE 0
                    END) * IF(c.dos_citas = 1, 2, 1)
            END
        )
    ) AS horas_restantes

FROM solicitud_voc s
LEFT JOIN cita c ON s.id_solicitud = c.id_solicitud
LEFT JOIN nota_cita nc ON c.id_cita = nc.id_cita

-- Filtro por ID de terapeuta
WHERE s.terapeuta = :idTerapeuta

GROUP BY 
    s.terapeuta,
    s.id_solicitud, 
    s.cliente, 
    s.num_sesiones

-- FILTRO POR CITAS DEL MES:
HAVING COUNT(
    CASE 
        WHEN YEAR(nc.fecha_creacion) = :anio AND MONTH(nc.fecha_creacion) = :mes 
        THEN nc.id_nota 
    END
) > 0;
